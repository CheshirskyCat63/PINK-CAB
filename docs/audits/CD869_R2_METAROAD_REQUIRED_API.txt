99: 	 *  arc length along the spline, matching FRoadPosition::SOffset. */

101: 	{
102: 		double S0 = 0.0;
103: 		double S1 = 0.0;
104: 		TInstancedStruct<FRoadZone> Zone;
105: 	};

107: 	/**
108: 	 * Build one span per FRoadLaneAttributeValueZone key on the lane: [LaneS0 + Key.SOffset, next key
109: 	 * or LaneS1), each carrying the lane zone already merged with that key's override. Spans are sorted by
110: 	 * S0 and never overlap; S before the first span keeps the lane's own zone.
111: 	 *
112: 	 * Attribute key SOffsets are LANE-RELATIVE (FRoadLane::Trim clips them to [0, End - Start]), hence the
113: 	 * LaneS0 rebase - the same one FProceduralPolygon_RoadLane does for the sidewalk-height attribute.
114: 	 *
115: 	 * Out is emptied first. It stays EMPTY when the lane has no such attribute, when no key falls inside
116: 	 * [LaneS0, LaneS1), or when the lane carries no zone at all - callers can then take their existing
117: 	 * whole-lane path unchanged.
118: 	 */

125: 	 */
