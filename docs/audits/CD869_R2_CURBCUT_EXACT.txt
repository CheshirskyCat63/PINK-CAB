/**
 * FRoadCurbCut â€” the geometry of ONE curb cut (a ramp where the sidewalk drops to road level), independent of
 * where it is authored.
 *
 * Two hosts use the same maths through a single NORMALIZED lateral coordinate, `Progress`:
 *   - a road LANE (FRoadLaneAttributeCurbCutValue): Progress = LaneAlpha / Alpha, i.e. the cut reaches
 *     `Alpha` of the lane width;
 *   - a closed RING fill (looped road zone / sidewalk spline): Progress = DistanceFromRing / Depth.
 * Progress 0 = at the curb (full cut), 1 = at the far edge of the cut (no cut), > 1 = outside the cut.
 *
 * The cut profile along S is a 4-point curve of cut STRENGTH (0 = untouched, 1 = fully lowered):
 *
 *     1.0        ______________
 *               /              \
 *     0.0  ____/                \____
 *          | F |     Width      | F |
 *
 * with Width/Falloff lerped from the (A) pair at Progress 0 to the (B) pair at Progress 1, so the cut can taper
 * as it goes inward. `bIsCross` pins the profile to the (A) pair at full strength across the whole extent â€” a
 * pedestrian-crossing style cut rather than a driveway ramp.
 *
 * Pure data + pure maths, in the RUNTIME module, so the editor attribute value AND the runtime sidewalk-spline
 * per-node settings can both drive it. The attribute keeps its own (identical) UPROPERTY block for
 * serialization compatibility with existing content and converts through FRoadLaneAttributeCurbCutValue::ToCurbCut().
 */
USTRUCT(BlueprintType)
struct METAROAD_API FRoadCurbCut
{
	GENERATED_BODY()

	/** Cross-walk style: full-strength cut of constant (A) width across the whole extent. Disables the B pair. */
	UPROPERTY(EditAnywhere, Category = CurbCut)
	bool bIsCross = false;

	/** Width of the fully-lowered part at the curb [cm]. */
	UPROPERTY(EditAnywhere, Category = CurbCut, meta = (UIMin = 10, ClampMin = 10, UIMax = 10000, ClampMax = 1000))
	double WidthA = 200;

	/** Ramp length added on each side of WidthA, at the curb [cm]. */
	UPROPERTY(EditAnywhere, Category = CurbCut, meta = (UIMin = 10, ClampMin = 10, UIMax = 1000, ClampMax = 1000))
	double FalloffA = 50;

	/** Width of the fully-lowered part at the far edge of the cut [cm]. */
	UPROPERTY(EditAnywhere, Category = CurbCut, meta = (UIMin = 10, ClampMin = 10, UIMax = 10000, ClampMax = 1000, EditCondition = "!bIsCross"))
	double WidthB = 200;

	/** Ramp length at the far edge of the cut [cm]. */
	UPROPERTY(EditAnywhere, Category = CurbCut, meta = (UIMin = 10, ClampMin = 10, UIMax = 1000, ClampMax = 1000, EditCondition = "!bIsCross"))
	double FalloffB = 0;

	/** How far the cut reaches INWARD from the ring [cm]. The ring-host counterpart of the lane host's `Alpha`. */
	UPROPERTY(EditAnywhere, Category = CurbCut, meta = (UIMin = 10, ClampMin = 10, UIMax = 2000, ClampMax = 10000))
	double Depth = 200;

	/** Subdivisions per profile span â€” the topology density of the ramp. */
	UPROPERTY(EditAnywhere, Category = CurbCut, meta = (UIMin = 1, ClampMin = 1, UIMax = 10, ClampMax = 10))
	int MeshDensity = 4;

	/** Smooth (CIM_CurveAuto) instead of straight ramps. */
	UPROPERTY(EditAnywhere, Category = CurbCut)
	bool bSmooth = false;

	/** Widest S span the cut can touch (the larger end, falloffs and the border samples included) [cm]. */
	double GetSpan() const;

	/**
	 * Cut-strength profile against ABSOLUTE S, at the normalized lateral `Progress`.
	 * Empty curve (evaluates to 0 = no cut) once Progress leaves [0,1] on a non-cross cut.
	 */
	FInterpCurve<double> GetCurveByProgress(double SCenter, double Progress) const;

	/**
	 * S positions the host must sample so the ramp has topology: a border sample, then MeshDensity samples per
	 * profile span, then the profile's end + a border sample. Appends; emits nothing outside the cut.
	 * `SPeriod > 0` wraps the emitted values into [0, SPeriod) â€” a closed ring's seam.
	 */
	void GetAdditionalOffsets(double SCenter, double Progress, TArray<double>& OutS, double SPeriod = 0.0) const;

	/**
	 * The lowered height at (S, Progress): `(1 - Strength(S) * HeightMul) * InHeight`, where HeightMul fades the
	 * cut out laterally (full at Progress 0, none at Progress 1; constant for a cross cut).
	 * `SPeriod > 0` measures S - SCenter along the shortest arc of a closed ring, so a cut may straddle the seam.
	 */
	double ApplyHeight(double SCenter, double S, double Progress, double InHeight, double SPeriod = 0.0) const;
};

/**
 * Ring geometry shared by every closed-ring curb-cut host that has no spline frame of its own (sidewalk-spline
 * fills and their curbs). The looped ROAD zone does NOT use these: its ring is the reference line, so it reads
 * (S, distance-from-ring) straight out of the spline frame instead.
 *
 * All rings here are CLOSED and carry NO closing duplicate â€” vertex i connects to (i+1) % Num.
 */
namespace MetaRoad::CurbCutRing
{
	/** Cumulative arc length at each vertex (Out[0] == 0), plus the wrap step; Out has Points.Num() + 1 entries,
	 *  so Out.Last() is the ring's total length. */
	METAROAD_API void BuildArcLengths(const TArray<FVector>& Points, TArray<double>& OutArcs);

	/** +1 when the ring is counter-clockwise in XY (interior on the LEFT of travel), -1 when clockwise. */
	METAROAD_API double InwardSign(const TArray<FVector>& Points);

	/** Nearest point on the ring: its arc length and the planar distance to it. */
	METAROAD_API void ProjectOnRing(const TArray<FVector>& Points, const TArray<double>& Arcs,
		const FVector2D& Query, double& OutArc, double& OutDistance);

	/** Ring point at `Arc`, pushed `InwardDistance` cm toward the interior (0 = on the ring). */
	METAROAD_API FVector2D SampleInward(const TArray<FVector>& Points, const TArray<double>& Arcs,
		double Arc, double InwardDistance, double InInwardSign);

	/**
	 * Split ring segments so that a vertex exists at every arc in `TargetArcs`. Arcs closer than
	 * `MinSpacing` to an existing vertex are skipped (they would make a degenerate segment).
	 *
	 * Returns the OLD -> NEW index map (one entry per input vertex) so the caller can rebase every index that
	 * refers into the ring â€” FMagneticRing::NodeStart and FRoadPolygonData::MagneticSpans both do.
	 */
	METAROAD_API void InsertExtraSamples(TArray<FVector>& Points, const TArray<double>& TargetArcs,
		TArray<int32>& OutOldToNew, double MinSpacing = 1.0);

	/**
	 * The stretch of ring a single cut can reach: an OPEN chain from the last vertex before
	 * `ArcCentre - HalfSpan` to the first one after `ArcCentre + HalfSpan`, wrapping across the seam when it has
	 * to. `OutArcs` are ABSOLUTE and strictly increasing â€” a chain that wrapped carries arcs past the ring length,
	 * which is exactly what the cut's own wrap-aware evaluation expects.
	 *
	 * Exists so a cut can be evaluated against a SMALL polyline instead of the whole ring: the build applies every
	 * cut to every vertex of the scope, and projecting onto a thousand-vertex ring per vertex per cut is not
	 * affordable.
	 */
	METAROAD_API void ExtractWindow(const TArray<FVector>& Points, const TArray<double>& Arcs,
		double ArcCentre, double HalfSpan, TArray<FVector2D>& OutChain, TArray<double>& OutArcs);

	/** Nearest point on an OPEN chain (no wrap): its absolute arc and the planar distance to it. */
	METAROAD_API void ProjectOnChain(const TArray<FVector2D>& Chain, const TArray<double>& Arcs,
		const FVector2D& Query, double& OutArc, double& OutDistance);
}

static void RemovePointsFromBegin(TArray<FVector2D>& Vertices, double Distance)
{
	if (Vertices.Num() && Distance >= 0)
	{
		double Dt = 0;
		for (int i = 1; i < Vertices.Num(); ++i)
		{
			Dt += (Vertices[i] - Vertices[i - 1]).Length();
			if (Dt >= Distance)
			{
				Vertices.RemoveAt(0, i);
				return;
			}
		}
		Vertices.Empty();
	}
}

static void RemovePointsFromEnd(TArray<FVector2D>& Vertices, double Distance)
{
	if (Vertices.Num() && Distance > 0)
	{
		double Dt = 0;
		for (int i = Vertices.Num() - 2; i >= 0; --i)
		{
			Dt += (Vertices[i] - Vertices[i + 1]).Length();
			if (Dt >= Distance)
			{
				Vertices.RemoveAt(i + 1, Vertices.Num() - i - 1);
				return;
			}
		}
		Vertices.Empty();
	}
}

/**
 * Index of the polyline point coincident with Pt, or INDEX_NONE.
 *
 * Tolerance is 2 x the polyline snap distance: NormalizePolyline may have dropped our point in favour of one up
 * to DefSnapDistance away in S. Still far below any real vertex spacing, so it cannot grab a different vertex.
 *
 * Used by every cross edge a lane inserts (today: the zone seams). Such an edge must SNAP onto a vertex that is
 * already in the arrangement rather than re-sample GetRoadPosition -- see the comment in InsertZoneBoundaryEdges
 * for what re-sampling costs.
 */
static int32 FindPointOnLine2D(const TArray<FVector2D>& Line, const FVector2D& Pt)
{
	const double SnapDistSq = FMath::Square(2.0 * GeomUtils::DefSnapDistance);

	int32 Best = INDEX_NONE;
	double BestDistSq = SnapDistSq;
	for (int32 i = 0; i < Line.Num(); ++i)
	{
		const double DistSq = FVector2D::DistSquared(Line[i], Pt);
		if (DistSq <= BestDistSq)
		{
			BestDistSq = DistSq;
			Best = i;
		}
	}
	return Best;
}


// If the polygon is not looped, then it simply returns Info.Pos.SOffset,
// otherwise it is necessary to determine the SOffset on the seam (0 or SplineLength)
static double GetSOffset(const FTriInfo& TriInfo, int TID, int VID)
{
	auto& VertexInfo = TriInfo.GetVertexInfo(VID);

	if (VertexInfo.bLoopSeam)
	{
		auto* Poly = static_cast<const FProceduralPolygon_RoadLane*>(VertexInfo.Poly);

		double SplineLength = Poly->GetRoadSpline().SplineCurves.GetSplineLength();

		check(VertexInfo.Poly);
		auto& BasOpData = VertexInfo.Poly->Owner;

		const FVector ForwardVector = VertexInfo.Pos.Quat.GetForwardVector();
		auto& T = BasOpData.Triangles[TID];

		const FVector& V_A = BasOpData.Vertices3d[T.A].Vertex;
		const FVector& V_B = BasOpData.Vertices3d[T.B].Vertex;
		const FVector& V_C = BasOpData.Vertices3d[T.C].Vertex;

		bool bIsSeam_A = TriInfo.A->bLoopSeam;
		bool bIsSeam_B = TriInfo.B->bLoopSeam;
		bool bIsSeam_C = TriInfo.C->bLoopSeam;

		if (bIsSeam_A && bIsSeam_B && bIsSeam_C)
		{
			ensure(false);
			return VertexInfo.Pos.SOffset;
		}

		TOptional<FVector> V1;
		TOptional<FVector> V2;

		if (VertexInfo.VID == T.A)
		{
			if (!bIsSeam_B)
			{
				V1 = (V_B - V_A).GetSafeNormal();
			}
			if (!bIsSeam_C)
			{
				V2 = (V_C - V_A).GetSafeNormal();
			}
		}
		else if (VertexInfo.VID == T.B)
		{
			if (!bIsSeam_A)
			{
				V1 = (V_A - V_B).GetSafeNormal();
			}
			if (!bIsSeam_C)
			{
				V2 = (V_C - V_B).GetSafeNormal();
			}
		}
		else if (VertexInfo.VID == T.C)
		{
			if (!bIsSeam_A)
			{
				V1 = (V_A - V_C).GetSafeNormal();
			}
			if (!bIsSeam_B)
			{
				V2 = (V_B - V_C).GetSafeNormal();
			}
		}
		else
		{
			ensure(false);
			return VertexInfo.Pos.SOffset;
		}

		if (V1.IsSet() && V2.IsSet())
		{
			if (ForwardVector.Dot(*V1) < 0 && ForwardVector.Dot(*V2) < 0)
			{
				return SplineLength;
			}
			else
			{
				return VertexInfo.Pos.SOffset;
			}
		}
		else if (V1.IsSet())
		{
			if (ForwardVector.Dot(*V1) < 0)
			{
				return SplineLength;
			}
			else
			{
				return VertexInfo.Pos.SOffset;
			}
		}
		else if (V2.IsSet())
		{
			if (ForwardVector.Dot(*V2) < 0)
			{
				return SplineLength;
			}
			else
			{
				return VertexInfo.Pos.SOffset;
			}
		}
		else
		{
			return VertexInfo.Pos.SOffset;
		}
	}
	else
	{
		return VertexInfo.Pos.SOffset;
	}
}

static TArray<FVector2D> GetPolyline2D(const TArray<FRoadPosition>& InPoints)
{
	if (InPoints.Num() == 0)
	{
		return {};
	}

	TArray<FVector2D> Points2D;
	Points2D.Reserve(InPoints.Num());
	for (auto& It : InPoints)
	{
		Points2D.Add(FVector2D{ It.Location });
	}
	GeomUtils::RemovedPolylineSelfIntersection(Points2D);
	return Points2D;
}

// Zone -> surface GID for a plain filling polygon (RoadLoop / Simple): Driving surface, else a Sidewalk maps to the


FProceduralPolygon_RoadLane::FProceduralPolygon_RoadLane(const FRoadTriangulationData& Owner, FRoadArrangementBuilder& Builder, int SplineIndex, int SectionIndex, int LaneIndex, double MaxSquareDistanceFromSpline, double MaxSquareDistanceFromCap, double MinSegmentLength)
	: FProceduralPolygon_RoadBase(Owner, SplineIndex)
	, SectionIndex(SectionIndex)
	, LaneIndex(LaneIndex)
	, MaxSquareDistanceFromSpline(MaxSquareDistanceFromSpline)
	, MinSegmentLength(MinSegmentLength)
	, bIsLoop(false)
{
	ResultInfo = { EGeometryResultType::InProgress };

	// Before anything else: the zone spans decide where the boundary polylines need a vertex and where a
	// constrained edge has to cross the lane. The centre line has no lane zone to override.
	if (LaneIndex != MetaRoad::ZeroLaneIndex)
	{
		MetaRoad::BuildRoadLaneZoneSpans(GetLane(), GetStartOffset(), GetEndOffset(), ZoneSpans);
	}

	// Shape, then insertion, then hooks. The order between them is the ORDER OF INSERTION into the arrangement,
	// which is byte-observable downstream (it numbers the vertices); the split into three is what makes that
	// order readable instead of implied by 200 lines of position in a constructor.
	const FLaneContour Contour = BuildLaneContour(MaxSquareDistanceFromCap);

	if (!InsertLaneContour(Builder, Contour))
	{
		return; // InsertLaneContour has already named which line failed in ResultInfo
	}

	RunLaneAttributeHooks(Builder);

	ResultInfo.SetSuccess();
}

FProceduralPolygon_RoadLane::FLaneContour FProceduralPolygon_RoadLane::BuildLaneContour(double MaxSquareDistanceFromCap) const
{
	// PURE: it samples the spline and reads the layout, and touches neither the arrangement nor Owner's mutable
	// state. Nothing here can half-build the polygon, because nothing here is stored on it.
	FLaneContour Out;

	const auto& RoadSplineCache = GetRoadSpline();
	const auto& Section = GetSection();
	const FRoadZoneSidewalk* AsSidewalk = GetRoadZone().GetPtr<FRoadZoneSidewalk>();

	// Taken from the INNER side because it is the one every lane has (the centre line included).
	const TArray<FRoadPosition> InsideLinePositions = GetPolyline(ERoadLaneSide::Inner);
	Out.Inside = GetPolyline2D(InsideLinePositions);

	if (LaneIndex != MetaRoad::ZeroLaneIndex)
	{
		Out.Outside = GetPolyline2D(GetPolyline(ERoadLaneSide::Outer));
	}

	if (AsSidewalk && LaneIndex != MetaRoad::ZeroLaneIndex)
	{
		const FRoadLane& Lane = GetLane();

		if (AsSidewalk->bEndCurb && AsSidewalk->EndCapCurve.GetRichCurveConst()->GetNumKeys())
		{
			double LaneWidth = Lane.Width.Eval(Lane.GetEndOffset());

			TArray<float> Values;
			TArray<float> Times;
			CurveUtils::CurveToPolyline(*AsSidewalk->EndCapCurve.GetRichCurveConst(), 0.0, 1.0, MaxSquareDistanceFromCap / (LaneWidth * LaneWidth), 0.0001f, 200, Values, Times);
			double MaxValue = *Algo::MaxElement(Values);

			for (int i = 0; i < Values.Num(); ++i)
			{
				Out.EndCap.Add(FVector2D{ RoadSplineCache.GetRoadPosition(SectionIndex, LaneIndex, Times[i], Lane.GetEndOffset() - (MaxValue - Values[i]) * LaneWidth, ESplineCoordinateSpace::World).Location});
			}
			if (MaxValue > 0)
			{
				RemovePointsFromEnd(Out.Inside, MaxValue * LaneWidth);
				RemovePointsFromEnd(Out.Outside, MaxValue * LaneWidth);

				Out.Inside.Add(Out.EndCap[0]);
				Out.Outside.Add(Out.EndCap.Last());
			}
		}
		if (AsSidewalk->bBeginCurb && AsSidewalk->BeginCapCurve.GetRichCurveConst()->GetNumKeys())
		{
			double LaneWidth = Lane.Width.Eval(Section.SOffset);

			TArray<float> Values;
			TArray<float> Times;
			CurveUtils::CurveToPolyline(*AsSidewalk->BeginCapCurve.GetRichCurveConst(), 0.0, 1.0, MaxSquareDistanceFromCap / (LaneWidth * LaneWidth), 0.0001f, 200, Values, Times);
			double MaxValue = *Algo::MaxElement(Values);

			for (int i = 0; i < Values.Num(); ++i)
			{
				Out.BeginCap.Add(FVector2D{ RoadSplineCache.GetRoadPosition(SectionIndex, LaneIndex, Times[i], Section.SOffset + (MaxValue - Values[i]) * LaneWidth, ESplineCoordinateSpace::World).Location });
			}
			if (MaxValue > 0)
			{
				RemovePointsFromBegin(Out.Inside, MaxValue * LaneWidth);
				RemovePointsFromBegin(Out.Outside, MaxValue * LaneWidth);

				Out.Inside.Insert(Out.BeginCap[0], 0);
				Out.Outside.Insert(Out.BeginCap.Last(), 0);
			}
		}
	}

	if (LaneIndex != MetaRoad::ZeroLaneIndex)
	{
		if (Out.BeginCap.Num() < 2)
		{
			Out.BeginCap.Empty();
			Out.BeginCap.Add(Out.Inside[0]);
			Out.BeginCap.Add(Out.Outside[0]);
		}

		if (Out.EndCap.Num() < 2)
		{
			Out.EndCap.Empty();
			Out.EndCap.Add(Out.Inside.Last());
			Out.EndCap.Add(Out.Outside.Last());
		}
	}

	if (LaneIndex == 0)
	{
		Out.GID = GUIFlags::CenterLine;
	}
	else if (GetRoadZone().GetPtr<FRoadZoneDriving>() != nullptr)
	{
		Out.GID = GUIFlags::DrivingSurface;
	}
	else if (AsSidewalk)
	{
		Out.GID = AsSidewalk->bIsSoftBorder ? GUIFlags::SidewalksSoft : GUIFlags::SidewalksHard;
	}

	return Out;
}

bool FProceduralPolygon_RoadLane::InsertLaneContour(FRoadArrangementBuilder& Builder, const FLaneContour& Contour)
{
	// THE ONE PHASE THAT TOUCHES THE OPEN ARRANGEMENT, and the order below is load-bearing: inside -> outside ->
	// endcap -> begincap -> zone seams. Vertices are numbered in insertion order, so a permutation here silently
	// renumbers every index the rest of the build derives from the graph.
	//
	// The five failure exits stay distinct: which line failed to insert is the only diagnostic this path produces,
	// and a lane that fails here fails the whole build in AddLanePolygons anyway.
	InsideLineInfo = Builder.InsertPolyline(Contour.Inside, Contour.GID);
	if (!InsideLineInfo.IsValid())
	{
		ResultInfo.SetFailed(FText::Format(LOCTEXT("FProceduralPolygon_RoadLane_InsideLineFailed", "FProceduralPolygon_RoadLane: {0}: InsideLineInfo failed"), GetDescription()));
		return false;
	}

	bIsLoop = InsideLineInfo.IsLoop();

	if (LaneIndex == MetaRoad::ZeroLaneIndex)
	{
		return true; // the centre line is a bare construction line: one polyline, no caps, no seams
	}

	OutsideLineInfo = Builder.InsertPolyline(Contour.Outside, Contour.GID);
	if (!OutsideLineInfo.IsValid())
	{
		ResultInfo.SetFailed(FText::Format(LOCTEXT("FProceduralPolygon_RoadLane_OutsideLineFailed", "FProceduralPolygon_RoadLane: {0}: OutsideLineInfo failed "), GetDescription()));
		return false;
	}

	if (InsideLineInfo.IsLoop() ^ OutsideLineInfo.IsLoop())
	{
		// There should not be situations when only one of the lines is a loop.
		ResultInfo.SetFailed(FText::Format(LOCTEXT("FProceduralPolygon_RoadLane_LoopFailed", "FProceduralPolygon_RoadLane: {0}: Wrong loop"), GetDescription()));
		return false;
	}

	EndCapInfo = Builder.InsertPolyline(Contour.EndCap, Contour.GID);
	BeginCapInfo = Builder.InsertPolyline(Contour.BeginCap, Contour.GID);

	if (!BeginCapInfo.IsValid())
	{
		ResultInfo.SetFailed(FText::Format(LOCTEXT("FProceduralPolygon_RoadLane_BeginCapInfo", "FProceduralPolygon_RoadLane: {0}: BeginCapInfo line info failed "), GetDescription()));
		return false;
	}

	if (!EndCapInfo.IsValid())
	{
		ResultInfo.SetFailed(FText::Format(LOCTEXT("FProceduralPolygon_RoadLane_EndCapInfo", "FProceduralPolygon_RoadLane: {0}: EndCapInfo line info failed "), GetDescription()));
		return false;
	}

	// Folded into this phase rather than given one of its own: the seams' only ordering constraint is "after both
	// boundary polylines are final", which is here. A separate phase would add the rule "immediately after the
	// second one" and buy nothing.
	InsertZoneBoundaryEdges(Builder, Contour.Inside, Contour.Outside);

	return true;
}

void FProceduralPolygon_RoadLane::RunLaneAttributeHooks(FRoadArrangementBuilder& Builder)
{
	// LAST, on a COMPLETE polygon: the hooks read this polygon's geometry, so they cannot run before the contour
	// is inserted. Builder -- not *this* -- is the sink now; the polygon is the subject only.
	if (LaneIndex != MetaRoad::ZeroLaneIndex)
	{
		for (const auto& [Desc, Attribute] : GetLane().Attributes)
		{
			if (Attribute.IsChildOf<FRoadLaneAttributePolygonCustomizationValue>())
			{
				for (auto& Key : Attribute.Keys)
				{
					if (auto* Value = Key.GetValuePtr< FRoadLaneAttributePolygonCustomizationValue>())
					{
						Value->OnPolygonCreated(*this, Builder, Key.SOffset);
					}
				}
			}
		}
	}

#if METAROAD_PRO
	// The centre line takes part in this one (its attributes live on the section), which is why it sits outside
	// the lane-only block above rather than beside the hook it looks like.
	for (const auto& [Desc, Attribute] : GetLaneAttributes())
	{
		if (Attribute.IsChildOf<FRoadLaneAttributePolygonValue>())
		{
			for (auto& Key : Attribute.Keys)
			{
				if (auto* Value = Key.GetValuePtr< FRoadLaneAttributePolygonValue>())
				{
					const FRoadPosition RoadPose = GetRoadSpline().EvalAttributeAnchorPosition(SectionIndex, LaneIndex, GetStartOffset() + Key.SOffset, *Value, ESplineCoordinateSpace::World);
					const bool bIsForward = (LaneIndex == MetaRoad::ZeroLaneIndex) || GetRoadSpline().GetRoadLayout().Sections[SectionIndex].GetLaneByIndex(LaneIndex).IsForwardLane();
					const FTransform Transform(RoadPose.Quat * FRotator(0.0, bIsForward ? -90.0 : +90.0, 0.0).Quaternion(), RoadPose.Location, FVector(-1.0, 1.0, 1.0));
					// Appends to the build's working fill list, in lane order -- which is what fixes
					// FProceduralPolygon_Simple's index and therefore GetObjectIndex().
					Builder.AddSurfacePolygons(Value->BuildPolygons(Transform));
				}
			}
		}
	}
#endif
}

bool FProceduralPolygon_RoadLane::OnCompleteArrangement(FRoadCompletionSink& Sink)
{
	if (ResultInfo.HasFailed())
	{
		return false;
	}

	FRoadVertexTable& Vertices = Sink.Vertices;

	if (!ProcessPolyline(Vertices, InsideLineInfo, InsideLineVertices, /*bLoopSeam*/ false))
	{
		ResultInfo.SetFailed(FText::Format(LOCTEXT("FProceduralPolygon_RoadLane_InsideLineNotFound", "FProceduralPolygon_RoadLane: {0}: Inside line not found after arrangement"), GetDescription()));
		return false;
	}

	if (LaneIndex != MetaRoad::ZeroLaneIndex)
	{
		if (!ProcessPolyline(Vertices, OutsideLineInfo, OutsideLineVertices, /*bLoopSeam*/ false))
		{
			ResultInfo.SetFailed(FText::Format(LOCTEXT("FProceduralPolygon_RoadLane_OutsideLineNotFound", "FProceduralPolygon_RoadLane: {0}: Outside line not found after arrangement"), GetDescription()));
			return false;
		}

		if (!bIsLoop)
		{
			if (!ProcessPolyline(Vertices, BeginCapInfo, BeginCapVertices, /*bLoopSeam*/ false))
			{
				ResultInfo.SetFailed(FText::Format(LOCTEXT("FProceduralPolygon_RoadLane_BeginCapeNotFound", "FProceduralPolygon_RoadLane: {0}: BeginCap line not found after arrangement"), GetDescription()));
				return false;
			}

			if (!ProcessPolyline(Vertices, EndCapInfo, EndCapVertices, /*bLoopSeam*/ false))
			{
				ResultInfo.SetFailed(FText::Format(LOCTEXT("FProceduralPolygon_RoadLane_EndCapLineNotFound", "FProceduralPolygon_RoadLane: {0}: EndCap line not found after arrangement"), GetDescription()));
				return false;
			}
		}
		else
		{
			if (!ProcessPolyline(Vertices, BeginCapInfo, BeginCapVertices, /*bLoopSeam*/ true))
			{
				ResultInfo.SetFailed(FText::Format(LOCTEXT("FProceduralPolygon_RoadLane_LoopSeamNotFound", "FProceduralPolygon_RoadLoop: {0}: LoopSeam line not found after arrangement"), GetDescription()));
				return false;
			}
		}
	}

	if (LaneIndex == MetaRoad::ZeroLaneIndex)
	{
		for (int i = 0; i < InsideLineVertices.Num() - 1; ++i)
		{
			Boundary.Add({ InsideLineVertices[i], InsideLineVertices[i + 1] });
		}
	}
	else if (bIsLoop)
	{
		FAxisAlignedBox2d InsideBound;
		for (int i = 0; i < InsideLineVertices.Num() - 1; ++i)
		{
			const auto& It = Owner.Vertices3d[InsideLineVertices[i]].Infos[this];
			InsideBound.Contain(FVector2D{ It.Pos.Location });
		}
		FAxisAlignedBox2d OutsideBound;
		for (int i = 0; i < OutsideLineVertices.Num() - 1; ++i)
		{
			const auto& It = Owner.Vertices3d[OutsideLineVertices[i]].Infos[this];
			OutsideBound.Contain(FVector2D{ It.Pos.Location });
		}

		auto AddBoundaries = [this](const TArray<int> & InBoundary, const TArray<int>& InHole)
		{
			for (int i = 0; i < InBoundary.Num() - 1; ++i)
			{
				Boundary.Add({ InBoundary[i], InBoundary[i + 1] });
			}
			Holes.Add({});
			for (int i = 0; i < InHole.Num() - 1; ++i)
			{
				Holes[0].Add({ InHole[i], InHole[i + 1]});
			}
			if (InBoundary.Num())
			{
				LocalSplineBounds.Contain(FVector2D{
					GetRoadSpline().SplineCurves.GetSplineLength(),
					Owner.Vertices3d[InBoundary[0]].Infos[this].Pos.ROffset
				});
			}
		};

		if (InsideBound.Area() > OutsideBound.Area())
		{
			AddBoundaries(InsideLineVertices, OutsideLineVertices);
		}
		else 
		{
			AddBoundaries(OutsideLineVertices, InsideLineVertices);
		}
	}
	else
	{
		for (int i = 0; i < InsideLineVertices.Num() - 1; ++i)
		{
			Boundary.Add({ InsideLineVertices[i], InsideLineVertices[i + 1] });
		}
		for (int i = 0; i < EndCapVertices.Num() - 1; ++i)
		{
			Boundary.Add({ EndCapVertices[i], EndCapVertices[i + 1] });
		}
		for (int i = OutsideLineVertices.Num() - 1; i >= 1; --i)
		{
			Boundary.Add({ OutsideLineVertices[i], OutsideLineVertices[i - 1] });
		}
		for (int i = BeginCapVertices.Num() - 1; i >= 1; --i)
		{
			Boundary.Add({ BeginCapVertices[i], BeginCapVertices[i - 1] });
		}
	}

	auto MakePoly = [this](TArray<FIndex2i>& InBoundary)
	{
		TArray<FVector2d> Vertex2d;
		{
			const auto& FirstInfo = Owner.Vertices3d[InBoundary[0].A].Infos[this];
			Vertex2d.Add(FVector2D{ FirstInfo.Pos.Location });
			LocalSplineBounds.Contain(FVector2D{ FirstInfo.Pos.SOffset, FirstInfo.Pos.ROffset });
		}
		for (const auto& It : InBoundary)
		{
			const auto& Info = Owner.Vertices3d[It.B].Infos[this];
			Vertex2d.Add(FVector2D{ Info.Pos.Location });
			LocalSplineBounds.Contain(FVector2D{ Info.Pos.SOffset, Info.Pos.ROffset });
		}
		auto Poly = UE::Geometry::FPolygon2d(Vertex2d);

		if (!Poly.IsClockwise())
		{
			Algo::Reverse(InBoundary);
			for (auto& It : InBoundary)
			{
				It = FIndex2i{ It.B, It.A };
			}
		}

		return Poly;
	};


	Poly2d = MakePoly(Boundary);

	if (Holes.Num())
	{
		MakePoly(Holes[0]);
	}

	GetSplineBounds().Contain(LocalSplineBounds);

	return true;
}

bool FProceduralPolygon_RoadLane::ProcessPolyline(FRoadVertexTable& Vertices, const FLineInfo& LineInfo, TArray<int>& VIDs, bool bLoopSeam)
{
	if (!FindPolyline(Owner.GetGraph(), LineInfo, VIDs))
	{
		return false;
	}

	for (int i = 0; i < VIDs.Num(); ++i)
	{
		AddVertexInfo(Vertices, VIDs[i], bLoopSeam, false);
	}

	return true;
};

void FProceduralPolygon_RoadLane::AddVertexInfo(FRoadVertexTable& Vertices, int VID, bool bLoopSeam) const
{
	AddVertexInfo(Vertices, VID, bLoopSeam, true);
}

void FProceduralPolygon_RoadLane::AddVertexInfo(FRoadVertexTable& Vertices, int VID, bool bLoopSeam, bool bWithUV1) const
{
	check(Owner.GetGraph().IsVertex(VID));

	const auto& SplineBounds = GetSplineBounds();

	// Both alphas normalize a lateral position by a WIDTH, and both widths can legitimately be zero:
	//   - the lane width, wherever a lane tapers to a point (a slip lane opening out of nothing, a lay-by
	//     closing) -- the correct way to author a lane appearing mid-road, since it leaves no step in the kerb;
	//   - the road-wide R extent, on a degenerate/one-vertex spline.
	// Dividing there produced NaN and fed it straight into UV0/UV1 of the drive surface. Collapse to 0 instead:
	// at zero width the inner and outer edges ARE the same line, so 0 (the inner edge) is the honest answer.
	const double RoadWidth = SplineBounds.Max.Y - SplineBounds.Min.Y;
	auto NormalizeLateral = [](double Numerator, double Width)
	{
		return FMath::IsNearlyZero(Width) ? 0.0 : Numerator / Width;
	};

	if (auto* FoundVertex = Vertices[VID].Infos.Find(this))
	{
		// If found just update and return
		FoundVertex->Alpha1 = bWithUV1 ? NormalizeLateral(FoundVertex->Pos.ROffset - SplineBounds.Max.Y, RoadWidth) : 0.0; // Alpha1 = [0..1]
		FoundVertex->bLoopSeam = FoundVertex->bLoopSeam || bLoopSeam;
		return;
	}

	const auto& RoadSplineCache = GetRoadSpline();

	const auto V2d = Owner.GetGraph().GetVertex(VID);
	const auto Pos = Owner.Inputs.Splines[GetSplineIndex()]->UpRayIntersection(V2d);
	const double ROffset1 = GetSection().EvalLaneROffset( LaneIndex, Pos.SOffset, 0.0);
	const double ROffset2 = GetSection().EvalLaneROffset( LaneIndex, Pos.SOffset, 1.0);

	auto& VertexInfo = Vertices[VID].Infos.Add(this);
	VertexInfo.Poly = this;
	VertexInfo.Pos = Pos;
	VertexInfo.Pos.Location.X = V2d.X;
	VertexInfo.Pos.Location.Y = V2d.Y;
	VertexInfo.Alpha0 = NormalizeLateral(Pos.ROffset - ROffset1, ROffset2 - ROffset1);  // Alpha0 = [0..1]
	VertexInfo.Alpha1 = bWithUV1 ? NormalizeLateral(VertexInfo.Pos.ROffset - SplineBounds.Max.Y, RoadWidth) : 0.0; // Alpha1 = [0..1]
	VertexInfo.VID = VID;
	VertexInfo.bLoopSeam = bLoopSeam;
	VertexInfo.HeightOffset = 0;

	if (const FRoadZoneDriving* LaneDriving = GetRoadZone().GetPtr<FRoadZoneDriving>())
	{
		if (LaneDriving && LaneDriving->bInvertUV0)
		{
			VertexInfo.Alpha0 = 1.0 - VertexInfo.Alpha0;
		}
	}
	else if (const FRoadZoneSidewalk* LaneSidewalk = GetRoadZone().GetPtr<FRoadZoneSidewalk>())
	{
		VertexInfo.HeightOffset = LaneSidewalk->DefaultHeight;

		const FRoadLaneAttributeSidewalkHeightValue DefHeightValue(VertexInfo.HeightOffset);

		const auto& Lane = GetLane();

		for (const auto& [Desc, Attribute] : Lane.Attributes)
		{
			if (Attribute.IsChildOf<FRoadLaneAttributeSidewalkHeightValue>())
			{
				VertexInfo.HeightOffset = Attribute.Evaluate(VertexInfo.Pos.SOffset - GetStartOffset(), DefHeightValue).Height;
			}
		}

		for (const auto& [Desc, Attribute] : Lane.Attributes)
		{
			if (Attribute.IsChildOf<FRoadLaneAttributePolygonCustomizationValue>())
			{
				for (auto& Key : Attribute.Keys)
				{
					if (auto* Value = Key.GetValuePtr< FRoadLaneAttributePolygonCustomizationValue>())
					{
						VertexInfo.HeightOffset = Value->GetHeightOffset(*this, Key.SOffset, Pos.SOffset, VertexInfo.Alpha0, VertexInfo.HeightOffset);
					}
				}
			}
		}
	}
}

const TInstancedStruct<FRoadZone>& FProceduralPolygon_RoadLane::GetRoadZoneAt(double SOffset) const
{
	if (ZoneSpans.IsEmpty())
	{
		return GetRoadZone();
	}

	const int32 Index = MetaRoad::FindRoadLaneZoneSpan(ZoneSpans, SOffset);
	return (Index != INDEX_NONE) ? ZoneSpans[Index].Zone : GetRoadZone();
}

const TInstancedStruct<FRoadZone>& FProceduralPolygon_RoadLane::GetRoadZoneForTri(int TID) const
{
	// The empty case is the norm; keeping it ahead of FindTri is what makes a lane without the attribute
	// cost exactly what it did before (three map lookups per triangle per op is not free).
	if (ZoneSpans.IsEmpty())
	{
		return GetRoadZone();
	}

	const FTriInfo TriInfo = FindTri(TID);
	if (!TriInfo.IsValid())
	{
		return GetRoadZone();
	}

	// Mean S of the three corners. InsertZoneBoundaryEdges put a CONSTRAINED edge on every boundary, so no
	// claimed triangle straddles one and the mean always lands in the span the whole triangle belongs to.
	// GetSOffset (not Pos.SOffset) because a loop seam vertex reads as 0 or SplineLength depending on which
	// side of the seam the triangle sits.
	const double S = (GetSOffset(TriInfo, TID, TriInfo.A->VID)
		            + GetSOffset(TriInfo, TID, TriInfo.B->VID)
		            + GetSOffset(TriInfo, TID, TriInfo.C->VID)) / 3.0;

	return GetRoadZoneAt(S);
}

void FProceduralPolygon_RoadLane::GetDistinctRoadZones(TArray<const TInstancedStruct<FRoadZone>*>& Out) const
{
	Out.Reset();
	Out.Add(&GetRoadZone());
	for (const FRoadLaneZoneSpan& Span : ZoneSpans)
	{
		Out.Add(&Span.Zone);
	}
}

void FProceduralPolygon_RoadLane::InsertZoneBoundaryEdges(FRoadArrangementBuilder& Builder, const TArray<FVector2D>& InsideLine2D, const TArray<FVector2D>& OutsideLine2D)
{
	// Without a constrained edge at the boundary the Delaunay lets triangles straddle it, and a material /
	// UV-density change would then follow a zigzag of triangle edges instead of a straight cross-section.
	//
	// GID = Other + PID = -1 is the curb-cut "rung" idiom (FRoadLaneAttributeCurbCutValue::OnPolygonCreated):
	// an untracked edge never becomes a polygon boundary, so FindPolyline ignores it and this polygon still
	// claims the triangles on BOTH sides -- the lane stays ONE polygon that merely gained an interior seam.
	//
	// Both endpoints are SNAPPED ONTO an existing boundary vertex rather than re-sampled, and the span is
	// skipped when either side has none. Re-sampling GetRoadPosition() looks equivalent -- GetPolyline() put a
	// point at exactly this S -- but is not: a sidewalk with cap curves calls RemovePointsFromEnd/Begin, which
	// strip MaxValue * LaneWidth cm off each end of BOTH polylines, boundary points included. The outer end of
	// such a span has no neighbouring lane to re-supply the vertex, so a re-sampled edge would introduce a
	// vertex outside every polygon -- one with no FRoadVertexInfo, which fails the whole build at the height
	// pass ("Mesh is broken", RoadBaseOp). Snapping also absorbs the case where NormalizePolyline kept a
	// neighbouring sample instead of ours.
	auto& RoadSplineCache = GetRoadSpline();

	for (const FRoadLaneZoneSpan& Span : ZoneSpans)
	{
		// The first span may start at the lane start, where the begin cap already IS the cross edge.
		if (Span.S0 - GetStartOffset() < GeomUtils::DefSnapDistance)
		{
			continue;
		}

		const FVector2D Inner{ RoadSplineCache.GetRoadPosition(SectionIndex, LaneIndex, 0.0, Span.S0, ESplineCoordinateSpace::World).Location };
		const FVector2D Outer{ RoadSplineCache.GetRoadPosition(SectionIndex, LaneIndex, 1.0, Span.S0, ESplineCoordinateSpace::World).Location };

		const int32 InnerIndex = FindPointOnLine2D(InsideLine2D, Inner);
		const int32 OuterIndex = FindPointOnLine2D(OutsideLine2D, Outer);

		if (InnerIndex == INDEX_NONE || OuterIndex == INDEX_NONE)
		{
			// Truncated away by a sidewalk cap (or dropped as a self-intersection). The zone still applies to
			// the triangles there -- only the straight seam is lost, and the boundary follows triangle edges.
			continue;
		}

		Builder.AddConstraintEdge(InsideLine2D[InnerIndex], OutsideLine2D[OuterIndex]);
	}
}

bool FProceduralPolygon_RoadLane::SetUVLayers(FDynamicMesh3& Mesh, int TID) const
{
	check(Mesh.Attributes()->NumUVLayers() >= 2);

	const int UVMaxSize = Owner.Params.UVMaxSize;

	const double SplineLength = GetSplineBounds().Max.X - GetSplineBounds().Min.X;

	// Snap the tile count to a whole number so V closes on a CLOSED spline's seam (GetSOffset returns exactly
	// SplineLength there, hence an integer V). Clamped to >= 1: RoundToInt is 0 whenever SplineLength < 0.5/Scale
	// (under 200 cm for UV0, 500 cm for UV1) -- which collapsed the whole V axis, and Alpha1Mul with it, so a short
	// junction connector sampled a single texel. The length guard covers empty/degenerate spline bounds.
	auto Aligned = [SplineLength](double Scale)
	{
		return (SplineLength > UE_KINDA_SMALL_NUMBER)
			? double(FMath::Max<int64>(1, FMath::RoundToInt(SplineLength * Scale))) / SplineLength
			: Scale;
	};

	const double UV0ScaleFactorAligned = Aligned(Owner.Params.UV0ScaleFactor);
	const double UV1ScaleFactorAligned = Aligned(Owner.Params.UV1ScaleFactor);

	const double Alpha1Mul = GetSplineBounds().Extents().Y * UV1ScaleFactorAligned * 2.0;

	auto TriInfo = FindTri(TID);

	if (ensure(TriInfo.IsValid()))
	{
		const double SOffsetA = GetSOffset(TriInfo, TID, TriInfo.A->VID);
		const double SOffsetB = GetSOffset(TriInfo, TID, TriInfo.B->VID);
		const double SOffsetC = GetSOffset(TriInfo, TID, TriInfo.C->VID);

		// Resolved PER TRIANGLE: a RoadZone attribute key can change bOverrideUVDensity along S. With no
		// spans GetRoadZoneAt() returns the lane zone straight away, so this is the old whole-polygon value.
		const double Density = ResolveUVDensity(GetRoadZoneAt((SOffsetA + SOffsetB + SOffsetC) / 3.0), Owner.Params.DefaultUVDensity);
		// UV2 is the uniform-density layer. Tile snapping only buys a seamless wrap on a closed spline; on an open one
		// it is pure density error (up to 0.5/N -- 25% at N=2), so it is applied only where it pays for itself.
		const double UV2Factor = IsLoop() ? Aligned(Density) : Density;

		if (auto* UVLayer0 = Mesh.Attributes()->GetUVLayer(0))
		{
			int MinV = (int)(FMath::Min3(SOffsetA, SOffsetB, SOffsetC) * UV0ScaleFactorAligned);
			MinV =  MinV - MinV % UVMaxSize;

			UVLayer0->SetTriangle(TID, FIndex3i{
				UVLayer0->AppendElement(FVector2f(TriInfo.A->Alpha0, SOffsetA * UV0ScaleFactorAligned - MinV)),
				UVLayer0->AppendElement(FVector2f(TriInfo.B->Alpha0, SOffsetB * UV0ScaleFactorAligned - MinV)),
				UVLayer0->AppendElement(FVector2f(TriInfo.C->Alpha0, SOffsetC * UV0ScaleFactorAligned - MinV))
			});
		}

		if (auto* UVLayer1 = Mesh.Attributes()->GetUVLayer(1))
		{
			int MinV = (int)(FMath::Min3(SOffsetA, SOffsetB, SOffsetC) * UV1ScaleFactorAligned);
			MinV = MinV - MinV % UVMaxSize;

			UVLayer1->SetTriangle(TID, FIndex3i{
				UVLayer1->AppendElement(FVector2f((TriInfo.A->Alpha1 - 0.5) * Alpha1Mul + 0.5, SOffsetA * UV1ScaleFactorAligned - MinV)),
				UVLayer1->AppendElement(FVector2f((TriInfo.B->Alpha1 - 0.5) * Alpha1Mul + 0.5, SOffsetB * UV1ScaleFactorAligned - MinV)),
				UVLayer1->AppendElement(FVector2f((TriInfo.C->Alpha1 - 0.5) * Alpha1Mul + 0.5, SOffsetC * UV1ScaleFactorAligned - MinV))
			});
		}

		if (auto* UVLayer2 = Mesh.Attributes()->GetUVLayer(2))
		{
			int MinV = (int)(FMath::Min3(SOffsetA, SOffsetB, SOffsetC) * UV2Factor);
			MinV = MinV - MinV % UVMaxSize;

			// U: centimetres from the SPLINE's reference line, so it runs continuously across lane boundaries.
			// (It used to be Alpha2 -- centimetres from each LANE's inner edge, restarting at 0 on every lane,
			// which cut the tiling mid-texel at every lane edge.)
			UVLayer2->SetTriangle(TID, FIndex3i{
				UVLayer2->AppendElement(FVector2f(TriInfo.A->Pos.ROffset * Density, SOffsetA * UV2Factor - MinV)),
				UVLayer2->AppendElement(FVector2f(TriInfo.B->Pos.ROffset * Density, SOffsetB * UV2Factor - MinV)),
				UVLayer2->AppendElement(FVector2f(TriInfo.C->Pos.ROffset * Density, SOffsetC * UV2Factor - MinV))
			});
		}

		return true;
	}

	return false;
}


TArray<FRoadPosition> FProceduralPolygon_RoadLane::GetPolyline(int InLaneIndex, double Alpha) const
{
	auto& RoadSplineCache = GetRoadSpline();

	if (!RoadSplineCache.RoadLayout.Sections.IsValidIndex(SectionIndex))
	{
		return {};
	}

	const auto& Section = RoadSplineCache.RoadLayout.Sections[SectionIndex];

	if (InLaneIndex != MetaRoad::ZeroLaneIndex && !Section.CheckLaneIndex(InLaneIndex))
	{
		return {};
	}

	TArray<FRoadPosition> OutPoints;

	if (InLaneIndex != MetaRoad::ZeroLaneIndex)
	{
		const auto& Lane = Section.GetLaneByIndex(InLaneIndex);


		for (int i = Lane.GetStartSectionIndex(); i <= Lane.GetEndSectionIndex(); ++i)
		{
			TArray<FRoadPosition> TmpPoints;
			if (RoadSplineCache.ConvertSplineToPolyline(
				[this, InLaneIndex, &RoadSplineCache, Alpha](double SOffset, ESplineCoordinateSpace::Type CoordinateSpace)
				{
					return RoadSplineCache.GetRoadPosition(SectionIndex, InLaneIndex, Alpha, SOffset, CoordinateSpace);
				},
				ESplineCoordinateSpace::World, MaxSquareDistanceFromSpline, MinSegmentLength,
				RoadSplineCache.RoadLayout.Sections[i].SOffset, RoadSplineCache.RoadLayout.Sections[i].SOffsetEnd_Cached,
				{}, true, TmpPoints))
			{
				if (OutPoints.Num())
				{
					OutPoints.RemoveAt(OutPoints.Num() - 1);
				}
				OutPoints.Append(MoveTemp(TmpPoints));
			}
		}

		if (Lane.RoadZone.GetPtr<FRoadZoneSidewalk>())
		{
			for (const auto& [Desc, Attribute] : Lane.Attributes)
			{
				if (Attribute.IsChildOf<FRoadLaneAttributePolygonCustomizationValue>())
				{
					for (auto& Key : Attribute.Keys)
					{
						if (auto* Value = Key.GetValuePtr< FRoadLaneAttributePolygonCustomizationValue>())
						{
							TArray<double> Segments;
							Value->GetAdditionalSplinePoints(*this, Key.SOffset, Alpha, Segments);

							TArray<FRoadPosition> AttributePoints;
							AttributePoints.Reserve(Segments.Num());
							for (double SOffset : Segments)
							{
								AttributePoints.Add(RoadSplineCache.GetRoadPosition(SectionIndex, InLaneIndex, Alpha, SOffset, ESplineCoordinateSpace::World));
							}
							GeomUtils::InsertPolyline(OutPoints, AttributePoints);
						}
					}
				}
			}
		}

		// Zone boundaries of the RoadZone attribute must be real vertices on this edge, so the constrained
		// edge InsertZoneBoundaryEdges lays across the lane meets the boundary AT a vertex instead of a
		// T-junction somewhere along a chord.
		//
		// Deliberately NOT gated by zone subtype (unlike the sidewalk-only block above): a RoadZone key is
		// legal on any lane. And deliberately keyed off `Lane` / `InLaneIndex` rather than this polygon's own
		// lane -- when the shared boundary between two lanes is assembled, this runs once for each of them
		// (see GetPolyline(ERoadLaneSide)), so both sides sample the same S and the seam welds.
		{
			TArray<double> BoundaryOffsets;
			MetaRoad::GetRoadLaneZoneBoundaryOffsets(Lane, Lane.GetStartOffset(), Lane.GetEndOffset(), BoundaryOffsets);

			// Appended then normalized in one pass. NOT GeomUtils::InsertPolyline: with more than one point it
			// first RemoveSegmant()s everything between the first and last S, which would delete the sampled
			// curve between two boundaries.
			for (double BoundaryS : BoundaryOffsets)
			{
				OutPoints.Add(RoadSplineCache.GetRoadPosition(SectionIndex, InLaneIndex, Alpha, BoundaryS, ESplineCoordinateSpace::World));
			}

			if (BoundaryOffsets.Num())
			{
				GeomUtils::NormalizePolyline(OutPoints);
			}
		}
	}
	else
	{
		RoadSplineCache.ConvertSplineToPolyline(
			[this, &RoadSplineCache](double SOffset, ESplineCoordinateSpace::Type CoordinateSpace)
			{
				return RoadSplineCache.GetRoadPosition(SectionIndex, MetaRoad::ZeroLaneIndex, 0.0, SOffset, CoordinateSpace);
			},
			ESplineCoordinateSpace::World, MaxSquareDistanceFromSpline, MinSegmentLength,
			Section.SOffset, Section.SOffsetEnd_Cached,
			{}, true, OutPoints);

		// THE SAME points the looped zone's ring takes, INSERTED the same way. This is a second sampling of the
		// reference line -- the ring is the first -- and two densities for one curve is two chains that never
		// weld. Inserted rather than passed above, so one offset is one vertex on both hosts (see
		// InsertCentreLineRingCuts).
		TArray<TArray<double>> RingCuts;
		GatherCentreLineRingOffsets(RoadSplineCache, RoadSplineCache.GetSplineLength(), RingCuts);
		InsertCentreLineRingCuts(RoadSplineCache, SectionIndex, Section.SOffset, Section.SOffsetEnd_Cached,
			RingCuts, OutPoints);
	}

	return OutPoints;
}

TArray<FRoadPosition> FProceduralPolygon_RoadLane::GetCenterPolyline(double S0, double S1) const
{
	TArray<FRoadPosition> OutPoints;

	auto& RoadSplineCache = GetRoadSpline();
	auto& Sections = RoadSplineCache.RoadLayout.Sections;

	// SUSPECTED REDUNDANCY (verify at runtime before "fixing"): GetPolyline(0/±1, 0.0) below all sample the MEMBER
	// SectionIndex's S-range, NOT this loop's `Section`, so on an N-section road the loop appends N IDENTICAL copies
	// (NormalizePolyline + GetSubPolyline then clip to [S0,S1]). Likely wasteful, possibly harmless (arrangement
	// dedup), possibly a multi-section boundary defect. Needs a baked multi-section road to confirm before changing.
	for (auto& Section: Sections)
	{
		OutPoints.Append(GetPolyline(0, 0.0));

		if(Section.Left.Num())
		{
			OutPoints.Append(GetPolyline(-1, 0.0));
		}

		if (Section.Right.Num())
		{
			OutPoints.Append(GetPolyline(+1, 0.0));
		}
	}

	GeomUtils::NormalizePolyline(OutPoints);
	OutPoints = GeomUtils::GetSubPolyline(OutPoints, S0, S1);

	return OutPoints;
}

TArray<FRoadPosition> FProceduralPolygon_RoadLane::GetPolyline(ERoadLaneSide Side) const
{
	auto& RoadSplineCache = GetRoadSpline();

	const auto& Section = RoadSplineCache.RoadLayout.Sections[SectionIndex];

	if (LaneIndex == MetaRoad::ZeroLaneIndex)
	{
		return GetCenterPolyline(Section.SOffset, Section.SOffsetEnd_Cached);
	}

	const auto& Lane = Section.GetLaneByIndex(LaneIndex);
	const double Alpha = Side == ERoadLaneSide::Inner ? 0.0 : 1.0;
	const double S0 = Lane.GetStartOffset();
	const double S1 = Lane.GetEndOffset();

	TArray<FRoadPosition> OutPoints = GetPolyline(LaneIndex, Alpha);

	if (Side == ERoadLaneSide::Inner)
	{
		if (LaneIndex == 1 || LaneIndex == -1)
		{
			OutPoints.Append(GetCenterPolyline(S0, S1));
		}
		else if(LaneIndex < 0)
		{
			OutPoints.Append(GetPolyline(LaneIndex + 1, 1.0));
		}
		else if (LaneIndex > 0)
		{
			OutPoints.Append(GetPolyline(LaneIndex - 1, 1.0));
		}
	}
	else // if (Side == ERoadLaneSide::Outer)
	{
		if (LaneIndex < 0)
		{
			OutPoints.Append(GetPolyline(LaneIndex - 1, 0.0));
		}
		else if (LaneIndex > 0)
		{
			OutPoints.Append(GetPolyline(LaneIndex + 1, 0.0));
		}
	}

	GeomUtils::NormalizePolyline(OutPoints);
	OutPoints = GeomUtils::GetSubPolyline(OutPoints, S0, S1);

	return OutPoints;
}

		TArray<int> LineVertices;        // ordered boundary vertex list (loop / simple polygons)

	struct METAROADEDITOR_API FProceduralPolygon_Simple : public FProceduralPolygon
	{
		FProceduralPolygon_Simple(const FRoadTriangulationData& Owner, FRoadArrangementBuilder& Builder, int SimplePolygonIndex);

		virtual ERoadPolygonType GetType() const override  { return ERoadPolygonType::Simple; }
		virtual bool OnCompleteArrangement(FRoadCompletionSink& Sink) override;
		virtual void FillVertexInfo(FRoadVertexInfo& Info, int VID, FVector2d V2d) const override;
		virtual const TInstancedStruct<FRoadZone>& GetRoadZone() const override;
		virtual FText GetDescription() const override;

		virtual double GetMaterialPriority() const override;
		virtual int GetObjectIndex() const override;

	protected:
		/** Per-node sidewalk height at a boundary vertex's 2D position: project onto the source ring
		 *  (Owner.SurfacePolygons[SimplePolygonIndex].Vertices) and lerp the bracketing VertexHeights. Caller must
		 *  ensure VertexHeights is sized 1:1 with Vertices (>= 2). */
		double EvaluateHeightAt2D(const FVector2d& V2d) const;

		/** Validate and insert every inner contour of this fill, each with its OWN PID so OnCompleteArrangement can
		 *  recover it independently of the boundary. A ring that is degenerate, self-intersecting, not STRICTLY
		 *  inside OuterPoints2D, or overlapping an already-accepted hole is dropped with a warning -- the fill then
		 *  comes out solid there, which is wrong but valid, where letting it through is a hard build failure.
		 *  Ctor-time only, after the boundary insert and before InsertCurbCutContours. */
		void InsertHoleRings(FRoadArrangementBuilder& Builder, const TArray<FVector2D>& OuterPoints2D, int GID);

		/** Carve every curb cut's inner chain + rungs into the arrangement, and register each cut as a scope-wide
		 *  site. Ctor-time only. */
		void InsertCurbCutContours(FRoadArrangementBuilder& Builder, const TArray<FVector2D>& RingPoints2D);

		/** Hand one cut to FRoadTriangulationData::CurbCutSites, windowed to the stretch of ring it can reach. */
		void RegisterCurbCutSite(FRoadArrangementBuilder& Builder, const FRoadCurbCutPlacement& Placement) const;

		/** Cumulative arc length of the source ring (Num + 1 entries; last = total). Built in the ctor only when
		 *  the fill actually carries curb cuts — nothing else needs it. */
		TArray<double> RingArcs;

		/** A magnetized boundary span that MATCHED a real arrangement path at ctor time, so its own (approximate)
		 *  segments were skipped instead of inserted. OnCompleteArrangement tags Path with this polygon's PID and
		 *  FindPolyline then recovers the boundary along the real road-lane geometry.
		 *  A span that does NOT match is never recorded here -- its segments are inserted like any plain fill's, which
		 *  is what keeps every vertex owned. (Skipping a span we cannot tag would strand the fill's other inserted
		 *  vertices with no Infos and fail the whole build at the height pass.) */
		struct FMagneticSpanRec
		{
			int32 VidA = INDEX_NONE;
			int32 VidB = INDEX_NONE;
			FMagneticSpan Span;
			TArray<int> Path; // matched at ctor time; re-validated (and re-matched if a later insert split it) on complete
		};
		TArray<FMagneticSpanRec> MagneticSpanRecs;

		/** Resolve each span to a real arrangement path, recording the ones that matched into MagneticSpanRecs and
		 *  flagging their segments in OutSkipSeg (indexed like Points2D) so the caller inserts everything else. */
		//~ Takes no builder: matching only READS the arrangement (FindPinnedVertex / MatchPolyline), and reads go
		//~ through the const Owner like every other read.
		void MatchMagneticSpans(const TArray<FVector2D>& Points2D, const TArray<FMagneticSpan>& Spans, TBitArray<>& OutSkipSeg);

		FLineInfo LineInfo;

		/** One FLineInfo per hole ring that PASSED validation and was inserted, in insertion order. NOT 1:1 with
		 *  Owner.SurfacePolygons[i].Holes -- a rejected hole gets no entry, so index arithmetic between the two is
		 *  invalid by construction. */
		TArray<FLineInfo> HoleLineInfos;

		/** The cleaned 2D hole rings, parallel to HoleLineInfos. Kept from the ctor because InsertCurbCutContours
		 *  must reject a cut contour that reaches INTO a hole: such a point is a graph vertex this fill will not
		 *  claim, i.e. a vertex no polygon owns, which hard-fails the height pass with "Mesh is broken". */
		TArray<TArray<FVector2D>> HolePoints2D;

		int SimplePolygonIndex;
		GeomUtils::FPolygonIntersectApproximation IntersectApproximation;
	};

	/**
	 * FProceduralPolygon_FeatureLine — a FRoadFeatureLine turned into an EXACT arrangement VID chain (published to
	 * FeatureLineWeldedPaths) that the sweep ops walk so the mark/curb hugs the mesh edges. Two ways in, one recovery:
	 *   - HugMode == Weld:      INSERT the line as CenterLine-GID constraint edges (splitting whatever it crosses).
	 *   - HugMode == Magnetize: INSERT NOTHING -- pin the endpoints, MatchPolyline the line's approximation onto the
	 *                           real geometry underneath it, and TAG that existing-edge path (RevalidateAndTagPath) at
	 *                           completion. Open lines only (RoadBaseOp gates closed magnetize out -> drape).
	 * Both then recover the chain via FindPolyline over LineInfo.PID.
	 *
	 * An OPEN polyline: it builds an open Boundary (first.A != last.B) so IsPolyline()==true -> skipped by
	 * FindBoundaries (its GID is CenterLine) and by the triangle-claim loop; it never fills. Fills Infos in
	 * OnCompleteArrangement by projecting each arrangement vertex onto the line's own 3D vertices (Z + optional
	 * per-vertex height), enough to satisfy the >=1-Infos invariant. Its RoadZone is INVALID on purpose (treated as a
	 * decal), so it does NOT influence surface Z wherever a real lane/loop/simple vertex coexists -- no crease under
	 * the marking. (Magnetize tags EXISTING lane vertices, which already own a real surface Info, so the added decal
	 * Info is inert there.)
	 */
	struct METAROADEDITOR_API FProceduralPolygon_FeatureLine : public FProceduralPolygon
	{
		FProceduralPolygon_FeatureLine(const FRoadTriangulationData& Owner, FRoadArrangementBuilder& Builder, int FeatureLineIndex);

		virtual ERoadPolygonType GetType() const override { return ERoadPolygonType::FeatureLine; }
		virtual bool OnCompleteArrangement(FRoadCompletionSink& Sink) override;
		virtual void FillVertexInfo(FRoadVertexInfo& Info, int VID, FVector2d V2d) const override;
		virtual const TInstancedStruct<FRoadZone>& GetRoadZone() const override;
		virtual FText GetDescription() const override;
		virtual bool SetUVLayers(FDynamicMesh3& Mesh, int TID) const override { return false; } // never fills -> never UV-mapped
		virtual double GetMaterialPriority() const override { return 0.0; }
		virtual int GetObjectIndex() const override;

	private:
		/** Z at a 2D point ON the line: project onto Points2D, lerp bracketing PointsZ. Every inserted/crossing
		 *  vertex is collinear on some segment, so the projection is exact. */
		double EvaluateZAt2D(const FVector2d& V2d) const;

		int FeatureLineIndex;
		bool bClosedLine = false;   // a closed contour (e.g. the MarkProfile border of a box junction / island)
		FLineInfo LineInfo;
		TArray<FVector2D> Points2D; // XY of the line's own vertices (closing duplicate stripped for closed lines)
		TArray<double>    PointsZ;  // Location.Z (+ VertexHeights if present), 1:1 with Points2D

		// Magnetize path only (HugMode == Magnetize): the line inserts nothing; instead its endpoints are pinned and its
		// approximation is matched onto existing edges at ctor time, then tagged in OnCompleteArrangement.
		bool bMagnetize = false;
		int32 VidA = INDEX_NONE;    // pinned endpoint vertices (the matched path's ends)
		int32 VidB = INDEX_NONE;
		TArray<int> MatchedPath;    // VID path matched at ctor; re-validated + tagged on complete (see RevalidateAndTagPath)
	};
